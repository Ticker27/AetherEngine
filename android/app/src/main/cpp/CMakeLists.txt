project(aether CXX)
cmake_minimum_required(VERSION 3.22)

add_library(aether SHARED aether.cpp)
target_compile_features(aether PRIVATE cxx_std_17)
target_compile_options(aether PRIVATE -Wall -Wextra)
# 16 KB page alignment (Android 15+ / ตรงมาตรฐานเดียวกับ libflutter ของ engine)
target_link_options(aether PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,common-page-size=16384")
find_library(log-lib log)
target_link_libraries(aether ${log-lib})
